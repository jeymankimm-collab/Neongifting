datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

enum Role {
  USER
  CREATOR
  ADMIN
}

model User {
  id            String         @id @default(uuid())
  username      String         @unique
  email         String         @unique
  passwordHash  String         // Securely hashed with bcrypt
  avatarUrl     String?
  bio           String?
  role          Role           @default(USER)
  walletBalance Decimal        @default(0.00)
  isBanned      Boolean        @default(false)
  createdAt     DateTime       @default(now())

  videos        Video[]
  campaigns     Campaign[]
  votesGiven    Vote[]
  sentGifts     GiftTransaction[] @relation("SentGifts")
  receivedGifts GiftTransaction[] @relation("ReceivedGifts")
}

model Video {
  id           String    @id @default(uuid())
  userId       String
  videoUrl     String
  thumbnailUrl String?
  caption      String
  likesCount   Int       @default(0)
  createdAt    DateTime  @default(now())

  user         User      @relation(fields: [userId], references: [id], onDelete: Cascade)
}

model Campaign {
  id          String    @id @default(uuid())
  creatorId   String
  title       String
  description String
  totalVotes  Int       @default(0)
  isActive    Boolean   @default(true)
  createdAt   DateTime  @default(now())

  creator     User      @relation(fields: [creatorId], references: [id])
  votes       Vote[]
}

model Vote {
  id         String    @id @default(uuid())
  campaignId String
  voterId    String
  createdAt  DateTime  @default(now())

  campaign   Campaign  @relation(fields: [campaignId], references: [id])
  voter      User      @relation(fields: [voterId], references: [id])

  @@unique([campaignId, voterId]) // One vote per user per campaign
}

model GiftTransaction {
  id         String    @id @default(uuid())
  senderId   String
  receiverId String
  giftName   String
  coinAmount Decimal
  createdAt  DateTime  @default(now())

  sender     User      @relation("SentGifts", fields: [senderId], references: [id])
  receiver   User      @relation("ReceivedGifts", fields: [receiverId], references: [id])
}
